    ```python
    async def bench_one(cluster_id: str) -> dict:
        """One end-to-end compile, sliced. Returns ms per stage.

        Stages:
          keyframes — ffmpeg extraction (unchanged)
          parallel  — asyncio.gather(caption-writer ‖ angle-selector) wall-clock
          publish   — _publish_segment direct db call
        """
        out: dict = {"ok": False, "err": None}
        try:
            t0 = time.monotonic()
            frames = await extract_cluster_keyframes(cluster_id)
            t1 = time.monotonic()
            if not frames:
                raise RuntimeError("no keyframes extracted")

            caption_data, angle_data = await asyncio.gather(
                _run_caption_writer_with_vision(cluster_id),
                _run_angle_selector(cluster_id),
            )
            t2 = time.monotonic()

            seg_id = await _publish_segment(
                cluster_id=cluster_id,
                ordered_clip_ids=angle_data["clip_ids"],
                caption=caption_data["caption"],
                location=caption_data["location"],
            )
            t3 = time.monotonic()

            out.update(
                keyframe_ms=(t1 - t0) * 1000,
                parallel_ms=(t2 - t1) * 1000,
                publish_ms=(t3 - t2) * 1000,
                total_ms=(t3 - t0) * 1000,
                n_frames=len(frames),
                segment_id=seg_id,
                ok=True,
            )
        except Exception as e:
            out["err"] = f"{type(e).__name__}: {e}"
        return out
    ```
